247 template <typename FunctionLambda>
248 bool solve_impl(FunctionLambda &&function,
249 const Vector &x_ini,
250 Vector &x_sol) {
251#define CMD "Optimist::Optimizer::NelderMead::solve_impl(...): "
252
253 // Reset internal variables
254 this->reset_counters();
255 if (this->m_verbose) {
256 this->header();
257 }
258
259 // Coefficients
260 Scalar rho{this->m_rho}, chi{this->m_chi}, gamma{this->m_gamma},
261 sigma{this->m_sigma};
262 this->m_dim = static_cast<Integer>(x_ini.size());
263 if (this->m_adaptive) {
264 const Scalar n{static_cast<Scalar>(this->m_dim)};
265 rho = 1.0;
266 chi = 1.0 + 2.0 / n;
267 gamma = 0.75 - 1.0 / (2.0 * n);
268 sigma = this->m_dim > 1 ? 1.0 - 1.0 / n : 0.5;
269 }
270
271 // Initialize simplex
272 auto eval = [this, &function](const Dense &x) -> Scalar {
273 Scalar f;
274 bool success{this->evaluate_function(function,
275 Utils::from_dense<Vector>(x),
276 f)};
277 OPTIMIST_ASSERT(success, CMD "function evaluation failed.");
278 return std::isfinite(f) ? f : INFTY;
279 };
280 this->initialize_simplex(eval, Utils::to_dense(x_ini), this->m_delta);
281
282 // Algorithm iterations
283 const Scalar tol{this->m_tolerance};
284 Move move{Move::INITIALIZE};
285 Dense centroid(this->m_dim), x_r(this->m_dim), x_e(this->m_dim),
286 x_c(this->m_dim);
287 for (this->m_iterations = 1;
288 this->m_iterations < this->m_max_iterations;
289 ++this->m_iterations) {
290 // Sort the simplex points: best (low), second worst (mid) and worst
291 // (high)
292 this->sort();
293 const Integer low{this->m_order.front()};
294 const Integer high{this->m_order.back()};
295 const Integer mid{this->m_order[this->m_dim - 1]};
296 const Scalar f_low{this->m_f(low)}, f_mid{this->m_f(mid)},
297 f_high{this->m_f(high)};
298
299 // Check convergence on both the function values and the simplex size
300 this->m_diameter = 0.0;
301 for (Integer i{0}; i <= this->m_dim; ++i) {
302 this->m_diameter =
303 std::max(this->m_diameter,
304 (this->m_p.col(i) - this->m_p.col(low)).norm());
305 }
306 const Scalar f_range{f_high - f_low};
307 if (this->m_verbose) {
308 this->info(f_range, move_name(move));
309 }
310 const Scalar x_scale{
311 static_cast<Scalar>(1.0) +
312 this->m_p.col(low).template lpNorm<Eigen::Infinity>()};
313 if (f_range <= tol * (1.0 + std::abs(f_low)) &&
314 this->m_diameter <= tol * x_scale) {
315 this->m_converged = true;
316 break;
317 }
318
319 // Restart the simplex if it degenerated (flat simplex)
320 if (this->m_dim > 1 && this->m_diameter > 0 &&
321 this->normalized_volume(low) < this->m_volume_tolerance) {
322 const Dense x_best(this->m_p.col(low));
323 this->initialize_simplex(eval, x_best, this->m_diameter);
324 move = Move::RESTART;
325 continue;
326 }
327
328 // Centroid of all points but the worst
329 centroid = (this->m_p.rowwise().sum() - this->m_p.col(high)) /
330 static_cast<Scalar>(this->m_dim);
331 const Dense direction{centroid - this->m_p.col(high)};
332
333 // Reflection
334 x_r = centroid + rho * direction;
335 const Scalar f_r{eval(x_r)};
336 if (f_r < f_low) {
337 // Expansion
338 x_e = centroid + rho * chi * direction;
339 const Scalar f_e{eval(x_e)};
340 if (f_e < f_r) {
341 this->replace(high, x_e, f_e);
342 move = Move::EXPAND_E;
343 } else {
344 this->replace(high, x_r, f_r);
345 move = Move::EXPAND_R;
346 }
347 } else if (f_r < f_mid) {
348 this->replace(high, x_r, f_r);
349 move = Move::REFLECT;
350 } else if (f_r < f_high) {
351 // Outside contraction
352 x_c = centroid + gamma * rho * direction;
353 const Scalar f_c{eval(x_c)};
354 if (f_c <= f_r) {
355 this->replace(high, x_c, f_c);
356 move = Move::CONTRACT_O;
357 } else {
358 this->shrink(eval, low, sigma);
359 move = Move::SHRINK;
360 }
361 } else {
362 // Inside contraction
363 x_c = centroid - gamma * direction;
364 const Scalar f_c{eval(x_c)};
365 if (f_c < f_high) {
366 this->replace(high, x_c, f_c);
367 move = Move::CONTRACT_I;
368 } else {
369 this->shrink(eval, low, sigma);
370 move = Move::SHRINK;
371 }
372 }
373 }
374
375 // Print bottom
376 if (this->m_verbose) {
377 this->bottom();
378 }
379
380 // Convergence data
381 this->sort();
382 x_sol = Utils::from_dense<Vector>(this->m_p.col(this->m_order.front()));
383 return this->m_converged;
384
385#undef CMD
386 }

392 static std::string move_name(const Move move) {
393 switch (move) {
394 case Move::INITIALIZE:

396 case Move::REFLECT:

398 case Move::EXPAND_E:
399 return "Expand";

403 return "Contract (outside)";
404 case Move::CONTRACT_I:

407 return "Shrink";
408 case Move::RESTART:

410 default:
411 return "Unknown";
412 }

421 template <typename Evaluator>
422 void initialize_simplex(Evaluator &&eval,
423 const Dense &x,
424 const Scalar delta) {
425 const Integer n{this->m_dim};
426 this->m_p.resize(n, n + 1);
427 this->m_f.resize(n + 1);
428 this->m_p.col(0) = x;
429 this->m_f(0) = eval(x);
430 if (this->m_method == Method::SPENDLEY) {
431 // Regular simplex with edge length delta
432 const Scalar sn{static_cast<Scalar>(n)};
433 const Scalar t{std::sqrt(sn + 1)};
434 const Scalar s2{std::sqrt(static_cast<Scalar>(2.0))};
435 const Scalar p{delta * (sn - 1 + t) / (sn * s2)};
436 const Scalar q{delta * (t - 1) / (sn * s2)};
437 for (Integer i{0}; i < n; ++i) {
438 this->m_p.col(i + 1) = x.array() + q;
439 this->m_p(i, i + 1) = x(i) + p;
440 this->m_f(i + 1) = eval(this->m_p.col(i + 1));
441 }
442 } else {
443 // Axis-aligned simplex, choosing the best of +delta and -delta
444 for (Integer i{0}; i < n; ++i) {
445 Dense x_p(x), x_m(x);
446 x_p(i) += delta;
447 x_m(i) -= delta;
448 const Scalar f_p{eval(x_p)}, f_m{eval(x_m)};
449 if (f_p <= f_m) {
450 this->m_p.col(i + 1) = x_p;
451 this->m_f(i + 1) = f_p;
452 } else {
453 this->m_p.col(i + 1) = x_m;
454 this->m_f(i + 1) = f_m;
455 }
456 }
457 }
458 this->m_order.resize(n + 1);
459 std::iota(this->m_order.begin(), this->m_order.end(), Integer(0));
460 }
461
465 void sort() {
466 std::stable_sort(this->m_order.begin(),
467 this->m_order.end(),
468 [this](const Integer a, const Integer b) {
469 return this->m_f(a) < this->m_f(b);
470 });
471 }
472
476 void replace(const Integer j, const Dense &x, const Scalar f) {
477 this->m_p.col(j) = x;
478 this->m_f(j) = f;
479 }

484 template <typename Evaluator>
485 void shrink(Evaluator &&eval, const Integer low, const Scalar sigma) {
486 for (Integer i{0}; i <= this->m_dim; ++i) {
487 if (i != low) {
488 this->m_p.col(i) =
489 this->m_p.col(low) +
490 sigma * (this->m_p.col(i) - this->m_p.col(low));
491 this->m_f(i) = eval(this->m_p.col(i));
492 }
493 }
494 }
495
501 Scalar normalized_volume(const Integer low) const {
502 const Integer n{this->m_dim};
503 DenseMatrix edges(n, n);
504 Integer j{0};
505 for (Integer i{0}; i <= n; ++i) {
506 if (i != low) {
507 edges.col(j++) = (this->m_p.col(i) - this->m_p.col(low)) /
508 this->m_diameter;
509 }
510 }
511 // Regular simplex with unit edge: sqrt(n+1) / (n! * 2^(n/2)), the
512 // factorial cancels out in the ratio
513 const Scalar sn{static_cast<Scalar>(n)};
514 const Scalar regular{std::sqrt(sn + 1) /
515 std::pow(static_cast<Scalar>(2.0), sn / 2)};
516 return std::abs(edges.determinant()) / regular;
517 }
