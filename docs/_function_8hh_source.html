191 bool is_solution(const Input &x,
192 const Scalar tol = FunctionBase::SQRT_EPSILON) const {
193 for (const auto &s : this->m_solutions) {
194 if constexpr (InputTrait::IsEigen) {
195 if ((x - s).norm() < tol) {
196 return true;
197 }
198 } else {
199 if (std::abs(x - s) < tol) {
200 return true;
201 }
