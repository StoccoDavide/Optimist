133 template <typename FunctionLambda,
134 typename GradientLambda,
135 typename HessianLambda>
136 bool solve_impl(FunctionLambda &&function,
137 GradientLambda &&gradient,
138 HessianLambda &&hessian,
139 const Vector &x_ini,
140 Vector &x_sol) {
141#define CMD "Optimist::Optimizer::TrustRegion::solve_impl(...): "
142
143 this->reset_counters();
144 if (this->m_verbose) {
145 this->header();
146 }
147
148 const Integer n{static_cast<Integer>(x_ini.size())};
149 Dense x(Utils::to_dense(x_ini)), g(n), p(n);
150 DenseMatrix B(n, n);
151 Scalar f, delta{this->m_radius};
152
153 auto eval_f = [this, &function](const Dense &z, Scalar &out) -> bool {
154 return this->evaluate_function(function,
155 Utils::from_dense<Vector>(z),
156 out) &&
157 std::isfinite(out);
158 };
159 auto eval_derivatives = [this, &gradient, &hessian](
160 const Dense &z,
161 Dense &g_out,
162 DenseMatrix &B_out) -> bool {
163 const Vector zv{Utils::from_dense<Vector>(z)};
164 FirstDerivative gv;
165 SecondDerivative Bv;
166 bool ok{this->evaluate_gradient(gradient, zv, gv) &&
167 this->evaluate_hessian(hessian, zv, Bv)};
168 if (!ok) {
169 return false;
170 }
171 g_out = Utils::to_dense(gv);
172 B_out = DenseMatrix(Bv);
173 return g_out.allFinite() && B_out.allFinite();
174 };
175
176 OPTIMIST_ASSERT(eval_f(x, f) && eval_derivatives(x, g, B),
177 CMD "evaluation failed at the initial point.");
178
179 std::string notes{"Initialize"};
180 for (this->m_iterations = 1;
181 this->m_iterations < this->m_max_iterations;
182 ++this->m_iterations) {
183 const Scalar g_norm{g.norm()};
184 if (this->m_verbose) {
185 this->info(g_norm, notes);
186 }
187 if (g_norm < this->m_tolerance) {
188 this->m_converged = true;
189 break;
190 }
191 if (delta < EPSILON * (1 + x.norm())) {
192 notes = "Radius collapsed";
193 break;
194 }
195
196 // Approximately solve min m(p) = g'p + p'Bp/2, |p| <= delta
197 if (this->m_method == Method::DOGLEG) {
198 this->dogleg(g, B, delta, p);
199 } else {
200 this->steihaug(g, B, delta, p);
201 }
202 const Scalar predicted{-(g.dot(p) + Scalar(0.5) * p.dot(B * p))};
203
204 // Actual vs predicted reduction
205 Dense x_trial{x + p};
206 Scalar f_trial;
207 Scalar rho{-1};
208 bool ok{false};
209 try {
210 ok = eval_f(x_trial, f_trial);
211 } catch (const std::runtime_error &) {
212 notes = "Budget exhausted";
213 break;
214 }
215 if (ok && predicted > 0) {
216 rho = (f - f_trial) / predicted;
217 }
218
219 // Radius update
220 const Scalar p_norm{p.norm()};
221 if (rho < 0.25) {
222 delta = 0.25 * p_norm;
223 notes = "Shrink";
224 } else if (rho > 0.75 && p_norm >= 0.99 * delta) {
225 delta = std::min(2 * delta, this->m_max_radius);
226 notes = "Expand";
227 } else {
228 notes = "Keep";
229 }
230
231 // Step acceptance
232 if (rho > this->m_eta) {
233 x = x_trial;
234 f = f_trial;
235 try {
236 OPTIMIST_ASSERT(eval_derivatives(x, g, B),
237 CMD "derivative evaluation failed.");
238 } catch (const std::runtime_error &) {
239 notes = "Budget exhausted";
240 break;
241 }
242 } else {
243 notes = "Reject";
244 }
245 }
246
247 if (this->m_verbose) {
248 this->bottom();
249 }
250 x_sol = Utils::from_dense<Vector>(x);
251 return this->m_converged;
252
253#undef CMD
254 }

261 static Scalar to_boundary(const Dense &z,
262 const Dense &d,
263 const Scalar delta) {
264 const Scalar a{d.squaredNorm()}, b{2 * z.dot(d)},
265 c{z.squaredNorm() - delta * delta};
266 const Scalar disc{std::max(Scalar(0), b * b - 4 * a * c)};
267 return (-b + std::sqrt(disc)) / (2 * a);
268 }
269
273 static void steihaug(const Dense &g,
274 const DenseMatrix &B,
275 const Scalar delta,
276 Dense &p) {
277 const Integer n{static_cast<Integer>(g.size())};
278 const Scalar g_norm{g.norm()};
279 const Scalar eps{std::min(Scalar(0.5), std::sqrt(g_norm)) * g_norm};
280 Dense z(Dense::Zero(n)), r(g), d(-g);
281 for (Integer j{0}; j < 2 * n; ++j) {
282 const Dense Bd{B * d};
283 const Scalar dBd{d.dot(Bd)};
284 if (dBd <= 0) {
285 // Negative curvature: go to the boundary
286 p = z + to_boundary(z, d, delta) * d;
287 return;
288 }
289 const Scalar alpha{r.squaredNorm() / dBd};
290 const Dense z_new{z + alpha * d};
291 if (z_new.norm() >= delta) {
292 p = z + to_boundary(z, d, delta) * d;
293 return;
294 }
295 const Dense r_new{r + alpha * Bd};
296 if (r_new.norm() < eps) {
297 p = z_new;
298 return;
299 }
300 const Scalar beta{r_new.squaredNorm() / r.squaredNorm()};
301 d = -r_new + beta * d;
302 r = r_new;
303 z = z_new;
304 }
305 p = z;
306 }
307
311 static void dogleg(const Dense &g,
312 const DenseMatrix &B,
313 const Scalar delta,
314 Dense &p) {
315 const Eigen::LLT<DenseMatrix> llt(B);
316 if (llt.info() != Eigen::Success) {
317 steihaug(g, B, delta, p);
318 return;
319 }
320 const Dense p_newton{-llt.solve(g)};
321 if (p_newton.norm() <= delta) {
322 p = p_newton;
323 return;
324 }
325 const Dense p_cauchy{-(g.squaredNorm() / g.dot(B * g)) * g};
326 if (p_cauchy.norm() >= delta) {
327 p = (delta / p_cauchy.norm()) * p_cauchy;
328 return;
329 }
330 p = p_cauchy + to_boundary(p_cauchy, p_newton - p_cauchy, delta) *
331 (p_newton - p_cauchy);
332 }
