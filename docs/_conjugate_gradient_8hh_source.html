331 template <typename FunctionLambda, typename GradientLambda>
332 bool solve_impl(FunctionLambda &&function,
333 GradientLambda &&gradient,
334 const Vector &x_ini,
335 Vector &x_sol) {
336#define CMD "Optimist::Optimizer::ConjugateGradient::solve_impl(...): "
337
338 this->reset_counters();
339 if (this->m_verbose) {
340 this->header();
341 }
342
343 // Current point (x, f, g), search direction d and trial step
344 Vector x(x_ini), x_new(x_ini);
345 Scalar f{0}, f_new{0};
346 FirstDerivative g, g_new, d;
347 OPTIMIST_ASSERT(this->evaluate_function(function, x, f),
348 CMD "function evaluation failed at the initial point.");
349 OPTIMIST_ASSERT(this->evaluate_gradient(gradient, x, g),
350 CMD "gradient evaluation failed at the initial point.");
351 d = -g;
352 Scalar trial{this->clamp_step(this->m_initial_step)}, accepted{0};
353
354 std::string notes{"Initialize"};
355 for (this->m_iterations = 1;
356 this->m_iterations < this->m_max_iterations;
357 ++this->m_iterations) {
358 if (this->m_verbose) {
359 this->info(g.norm(), notes);
360 }
361 if (g.norm() < this->m_tolerance) {
362 this->m_converged = true;
363 break;
364 }
365 notes = "-";
366 if (!this->has_sufficient_descent(g, d)) {
367 d = -g;
368 notes = "Restart";
369 }
370
371 // Line search. On failure, or if the step stagnates below the
372 // rounding level of x, retry once along the steepest descent
373 // direction before giving up
374 const bool ok{this->line_search(function, gradient, x, f, g, d, trial,
375 x_new, f_new, g_new, accepted)};
376 const bool steepest{(d + g).norm() <= EPSILON * g.norm()};
377 const bool stagnated{ok && (x_new - x).norm() <=
378 EPSILON * (1 + x.norm())};
379 if (!ok || stagnated) {
380 if (steepest) {
381 notes = ok ? "Stagnation" : "Line search failed";
382 if (ok) {
383 x = x_new;
384 f = f_new;
385 g = g_new;
386 this->m_converged = g.norm() < this->m_tolerance;
387 }
388 break;
389 }
390 if (ok) {
391 x = x_new;
392 f = f_new;
393 g = g_new;
394 }
395 d = -g;
396 trial = this->clamp_step(this->m_initial_step);
397 notes = "Restart";
398 continue;
399 }
400
401 // New conjugate direction
402 const FirstDerivative y{g_new - g};
403 const Scalar beta{this->compute_beta(g, g_new, d, y)};
404 FirstDerivative d_new{-g_new + beta * d};
405 if (!std::isfinite(beta) || this->should_restart(g, g_new) ||
406 !this->has_sufficient_descent(g_new, d_new)) {
407 d_new = -g_new;
408 notes = "Restart";
409 }
410 trial = this->next_trial_step(accepted, d, g, g_new);
411 x = x_new;
412 f = f_new;
413 g = g_new;
414 d = d_new;
415 }
416
417 if (this->m_verbose) {
418 this->bottom();
419 }
420 x_sol = x;
421 return this->m_converged;
422
423#undef CMD
424 }

431 Scalar clamp_step(Scalar t_step) const {
432 if (!std::isfinite(t_step) || t_step <= 0) {
433 t_step = this->m_initial_step;
434 }
435 return std::clamp(t_step, this->m_min_step, this->m_settings.alpha_max);
436 }
437
441 Scalar next_trial_step(const Scalar accepted,
442 const FirstDerivative &d,
443 const FirstDerivative &g,
444 const FirstDerivative &g_new) const {
445 if (this->m_alpha_method == AlphaMethod::BARZILAI_BORWEIN) {
446 // alpha = s's / s'y along the previous direction, s = accepted * d
447 const FirstDerivative y{g_new - g};
448 const Scalar dy{d.dot(y)};
449 if (dy > EPSILON * d.norm() * y.norm()) {
450 return this->clamp_step(accepted * d.squaredNorm() / dy);
451 }
452 }
453 return this->clamp_step(accepted);
454 }
455
460 bool has_sufficient_descent(const FirstDerivative &g,
461 const FirstDerivative &d) const {
462 const Scalar gd{g.dot(d)};
463 return std::isfinite(gd) &&
464 gd <= -this->m_descent_parameter * g.squaredNorm();
465 }
466
470 bool should_restart(const FirstDerivative &g,
471 const FirstDerivative &g_new) const {
472 if (!this->m_powell_restart) {
473 return false;
474 }
475 const Scalar n{g.norm() * g_new.norm()};
476 return n > 0 && g_new.dot(g) <= -this->m_restart_tolerance * n;
477 }
478
492 Scalar compute_beta(const FirstDerivative &g,
493 const FirstDerivative &g_new,
494 const FirstDerivative &d,
495 const FirstDerivative &y) const {
496 const Scalar gg{g.squaredNorm()};
497 const Scalar gg_new{g_new.squaredNorm()};
498 const Scalar dg{-d.dot(g)};
499 const Scalar dy{d.dot(y)};
500 const Scalar gy{g_new.dot(y)};
501 const bool ok_gg{gg > 0};
502 const bool ok_dg{dg > EPSILON * d.norm() * g.norm()};
503 const bool ok_dy{std::abs(dy) > EPSILON * d.norm() * y.norm()};
504
505 // Hager-Zhang: g_new'(y - 2 d |y|^2/(d'y)) / (d'y)
506 auto hager_zhang = [&]() -> Scalar {
507 return (gy - 2 * y.squaredNorm() / dy * g_new.dot(d)) / dy;
508 };
509
510 switch (this->m_beta_method) {
511 case BetaMethod::FLETCHER_REEVES:
512 return ok_gg ? gg_new / gg : 0;
513 case BetaMethod::POLAK_RIBIERE:
514 return ok_gg ? gy / gg : 0;
515 case BetaMethod::POLAK_RIBIERE_PLUS:
516 return ok_gg ? std::max(gy / gg, Scalar(0)) : 0;
517 case BetaMethod::HESTENES_STIEFEL:
518 return ok_dy ? gy / dy : 0;
519 case BetaMethod::CONJUGATE_DESCENT:
520 return ok_dg ? gg_new / dg : 0;
521 case BetaMethod::LIU_STOREY:
522 return ok_dg ? gy / dg : 0;
523 case BetaMethod::DAI_YUAN:
524 return ok_dy ? gg_new / dy : 0;
525 case BetaMethod::HAGER_ZHANG:
526 return ok_dy ? hager_zhang() : 0;
527 case BetaMethod::HAGER_ZHANG_PLUS: {
528 if (!ok_dy) {
529 return 0;
530 }
531 // Truncation eta_k = -1/(|d| min(eta, |g_new|))
532 const Scalar den{d.norm() *
533 std::min(this->m_truncation_parameter,
534 std::sqrt(gg_new))};
535 const Scalar bound{den > 0 ? -1 / den : -INFTY};
536 return std::max(hager_zhang(), bound);
537 }
538 default:
539 return 0;
540 }

548 template <typename FunctionLambda, typename GradientLambda>
549 bool line_search(FunctionLambda &function,
550 GradientLambda &gradient,
551 const Vector &x,
552 const Scalar f,
553 const FirstDerivative &g,
554 const FirstDerivative &d,
555 const Scalar trial,
556 Vector &x_new,
557 Scalar &f_new,
558 FirstDerivative &g_new,
559 Scalar &accepted) {
560#define CMD "Optimist::Optimizer::ConjugateGradient::line_search(...): "
561
562 const Scalar slope{g.dot(d)};
563 OPTIMIST_ASSERT(slope < 0, CMD "non-descent search direction.");
564
565 // Restriction of the objective, keeping the last evaluated point
566 Scalar last{-1};
567 bool g_valid{false};
568 this->m_relaxations = 0;
569 auto phi = [&](const Scalar a, Scalar &v, Scalar *dv) -> bool {
570 ++this->m_relaxations;
571 last = a;
572 g_valid = false;
573 x_new = x + a * d;
574 if (!this->evaluate_function(function, x_new, v)) {
575 return false;
576 }
577 f_new = v;
578 if (dv != nullptr) {
579 if (!this->evaluate_gradient(gradient, x_new, g_new)) {
580 return false;
581 }
582 g_valid = true;
583 *dv = g_new.dot(d);
584 }
585 return true;
586 };
587
588 accepted = trial;
589 Scalar value{f}, dvalue{slope};
590 if (!LineSearch::search(this->m_line_search, phi, f, slope, accepted,
591 value, dvalue, this->m_settings)) {
592 return false;
593 }
594
595 // Make sure the outputs refer to the accepted step
596 if (last != accepted) {
597 x_new = x + accepted * d;
598 OPTIMIST_ASSERT(this->evaluate_function(function, x_new, f_new),
599 CMD "function evaluation failed.");
600 g_valid = false;
601 }
602 if (!g_valid) {
603 OPTIMIST_ASSERT(this->evaluate_gradient(gradient, x_new, g_new),
604 CMD "gradient evaluation failed.");
605 }
606 return true;
607
608#undef CMD
609 }
