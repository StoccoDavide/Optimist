189 bool first_derivative_impl(const Vector &x, FirstDerivative &out) const {
190 DenseMatrix J(M, N);
191 this->derived().jacobian(Utils::to_dense(x), J);
192 out = Utils::from_dense_matrix<FirstDerivative>(J);
193 return J.allFinite();
194 }

199 bool second_derivative_impl(const Vector &x,
200 SecondDerivative &out) const {
201 using Matrix = typename SecondDerivative::value_type;
202 const Dense xd(Utils::to_dense(x));
203 out.resize(M);
204 bool finite{true};
205 for (Integer i{0}; i < M; ++i) {
206 DenseMatrix H(N, N);
207 this->derived().hessian(xd, i, H);
208 finite = finite && H.allFinite();
209 out[i] = Utils::from_dense_matrix<Matrix>(H);
210 }
211 return finite;
212 }
