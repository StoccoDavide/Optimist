386 template <typename FunctionLambda>
387 bool solve_impl(FunctionLambda &&function,
388 const Vector &x_ini,
389 Vector &x_sol) {
390 this->reset_counters();
391 if (this->m_verbose) {
392 this->header();
393 }
394
395 const Integer n{static_cast<Integer>(x_ini.size())};
396 const Integer m{this->m_particles > 0
397 ? this->m_particles
398 : 10 + static_cast<Integer>(
399 2 * std::sqrt(static_cast<Scalar>(n)))};
400 const Dense x0(Utils::to_dense(x_ini));
401
402 // Search box
403 Dense lb(n), ub(n);
404 if (this->has_finite_bounds()) {
405 lb = Utils::to_dense(this->m_lower_bound);
406 ub = Utils::to_dense(this->m_upper_bound);
407 } else {
408 lb = x0.array() - this->m_radius;
409 ub = x0.array() + this->m_radius;
410 }
411 const Dense width{ub - lb};
412 const Dense v_max{this->m_max_velocity * width};
413
414 // Swarm initialization (the first particle is the initial guess)
415 std::uniform_real_distribution<Scalar> unif(0, 1);
416 DenseMatrix x(n, m), v(n, m), p(n, m);
417 Dense f(m), f_p(m);
418 for (Integer i{0}; i < m; ++i) {
419 for (Integer j{0}; j < n; ++j) {
420 x(j, i) = lb(j) + unif(this->m_generator) * width(j);
421 v(j, i) = (2 * unif(this->m_generator) - 1) * v_max(j);
422 }
423 }
424 x.col(0) = x0.cwiseMax(lb).cwiseMin(ub);
425 for (Integer i{0}; i < m; ++i) {
426 f(i) = this->eval(function, x.col(i));
427 }
428 p = x;
429 f_p = f;
430 Integer best;
431 f_p.minCoeff(&best);
432
433 std::string notes{"Initialize"};
434 for (this->m_iterations = 1;
435 this->m_iterations < this->m_max_iterations;
436 ++this->m_iterations) {
437 // Convergence: personal bests collapsed onto the global best
438 Scalar radius{0};
439 for (Integer i{0}; i < m; ++i) {
440 radius = std::max(radius, (p.col(i) - p.col(best)).norm());
441 }
442 if (this->m_verbose) {
443 this->info(radius, notes);
444 }
445 // Converged when the personal bests collapsed onto the global best,
446 // or when they are all equal to rounding level in value (the swarm
447 // cannot resolve the minimizer any further)
448 const Scalar scale{1 + p.col(best).norm()};
449 const Scalar spread{f_p.maxCoeff() - f_p(best)};
450 if (radius < this->m_tolerance * scale ||
451 (spread <= 100 * EPSILON * std::max(Scalar(1), std::abs(f_p(best))) &&
452 radius < std::sqrt(this->m_tolerance) * scale)) {
453 this->m_converged = true;
454 break;
455 }
456 if (this->m_function_evaluations + m >
457 this->m_max_function_evaluations) {
458 notes = "Budget exhausted";
459 break;
460 }
461
462 // Velocity and position update
463 const Scalar w{this->m_weight_strategy(this->m_iterations,
464 this->m_max_iterations)};
465 for (Integer i{0}; i < m; ++i) {
466 for (Integer j{0}; j < n; ++j) {
467 Scalar vel{w * v(j, i) +
468 this->m_phi_p * unif(this->m_generator) *
469 (p(j, i) - x(j, i)) +
470 this->m_phi_g * unif(this->m_generator) *
471 (p(j, best) - x(j, i))};
472 vel = std::clamp(vel, -v_max(j), v_max(j));
473 v(j, i) = vel;
474 x(j, i) = x(j, i) + vel;
475 // Reflect on the box walls
476 if (x(j, i) < lb(j)) {
477 x(j, i) = lb(j);
478 v(j, i) = -v(j, i) / 2;
479 } else if (x(j, i) > ub(j)) {
480 x(j, i) = ub(j);
481 v(j, i) = -v(j, i) / 2;
482 }
483 }
484 f(i) = this->eval(function, x.col(i));
485 if (f(i) < f_p(i)) {
486 f_p(i) = f(i);
487 p.col(i) = x.col(i);
488 }
489 }
490 const Integer best_old{best};
491 f_p.minCoeff(&best);
492 notes = best != best_old ? "New best" : "-";
493 }
494
495 if (this->m_verbose) {
496 this->bottom();
497 }
498 x_sol = Utils::from_dense<Vector>(p.col(best));
499 return this->m_converged;
500 }
