130 void reset_bounds(const Integer n = InputTrait::IsFixed
131 ? InputTrait::Dimension
132 : 0) {
133#define CMD "Optimist::Solver::reset_bounds(...): "
134
135 if constexpr (InputTrait::IsScalar) {
136 this->m_lower_bound = -INFTY;
137 this->m_upper_bound = +INFTY;
138 } else if constexpr (InputTrait::IsFixed) {
139 this->m_lower_bound.setConstant(-INFTY);
140 this->m_upper_bound.setConstant(+INFTY);
141 } else if constexpr (InputTrait::IsDynamic) {
142 this->m_lower_bound.resize(n);
143 this->m_lower_bound.setConstant(-INFTY);
144 this->m_upper_bound.resize(n);
145 this->m_upper_bound.setConstant(+INFTY);
146 } else if constexpr (InputTrait::IsSparse) {
147 this->m_lower_bound.resize(n);
148 this->m_lower_bound.reserve(n);
149 this->m_upper_bound.resize(n);
150 this->m_upper_bound.reserve(n);
151 for (Integer i{0}; i < n; ++i) {
152 this->m_lower_bound.coeffRef(i) = -INFTY;
153 this->m_upper_bound.coeffRef(i) = +INFTY;
154 }
155 } else {
156 OPTIMIST_ERROR(CMD "unsupported input type for bounds reset.");
157 }

159#undef CMD

254 static bool valid_bounds(const Input &t_lower, const Input &t_upper) {
255 if constexpr (InputTrait::IsScalar) {
256 return !(t_upper <= t_lower);
257 } else {
258 using Dense = Eigen::Matrix<Scalar, Eigen::Dynamic, 1>;
259 if (t_lower.size() != t_upper.size() || t_lower.size() == 0) {
260 return true;
261 }
262 const Dense lower(t_lower), upper(t_upper);
263 for (Eigen::Index i{0}; i < lower.size(); ++i) {
264 if (!(upper(i) > lower(i))) {
265 return false;
266 }
267 }
268 return true;
269 }

816 template <typename FunctionValue, typename FunctionJacobian, typename Out>
817 static void least_squares_gradient(const FunctionValue &f,
818 const FunctionJacobian &J,
819 Out &out) {
820 if constexpr (TypeTrait<FunctionValue>::IsScalar &&
821 TypeTrait<FunctionJacobian>::IsScalar) {
822 out = J * f;
823 } else if constexpr (TypeTrait<FunctionValue>::IsScalar) {
824 out = f * J;
825 } else {
826 out = J.transpose() * f;
827 }
828 }

835 template <typename FunctionValue,
836 typename FunctionJacobian,
837 typename FunctionHessian,
838 typename Out>
839 static void least_squares_hessian(const FunctionValue &f,
840 const FunctionJacobian &J,
841 const FunctionHessian &H,
842 Out &out) {
843 if constexpr (TypeTrait<FunctionValue>::IsScalar &&
844 TypeTrait<FunctionJacobian>::IsScalar) {
845 out = J * J + f * H;
846 } else if constexpr (TypeTrait<FunctionValue>::IsScalar) {
847 out = J * J.transpose() + f * H;
848 } else {
849 out = J.transpose() * J;
850 for (Integer i{0}; i < static_cast<Integer>(H.size()); ++i) {
851 Scalar f_i;
852 if constexpr (TypeTrait<FunctionValue>::IsSparse) {
853 f_i = f.coeff(i);
854 } else {
855 f_i = f(i);
856 }
857 if (f_i != 0.0) {
858 out += f_i * H[i];
859 }
860 }
861 }

875 template <Problem Kind,
876 typename FunctionInput,
877 typename FunctionOutput,
878 typename DerivedFunction>
879 bool solve(
880 const FunctionBase<FunctionInput, FunctionOutput, DerivedFunction>
881 &function,
882 const Input &x_ini,
883 Input &x_sol) {
884#define CMD "Optimist::Solver::solve(...): "
885
886 using FunctionType =
887 FunctionBase<FunctionInput, FunctionOutput, DerivedFunction>;
888 using FunctionFirstDerivative = typename FunctionType::FirstDerivative;
889 using FunctionSecondDerivative = typename FunctionType::SecondDerivative;
890
891 // Lambda generators for the function and its derivatives as seen by the
892 // solver implementation
893 auto function_lambda = [&function](const Input &x, Output &out) -> bool {
894 if constexpr (Kind == Problem::ROOT || Kind == Problem::COST) {
895 return function.evaluate(x, out);
896 } else if constexpr (Kind == Problem::LEAST_SQUARES) {
897 FunctionOutput f;
898 if (!function.evaluate(x, f)) {
899 return false;
900 }
901 if constexpr (TypeTrait<FunctionOutput>::IsScalar) {
902 out = 0.5 * f * f;
903 } else {
904 out = 0.5 * f.squaredNorm();
905 }
906 return std::isfinite(out);
907 } else if constexpr (Kind == Problem::STATIONARY_COST) {
908 return function.first_derivative(x, out);
909 } else {
910 FunctionOutput f;
911 FunctionFirstDerivative J;
912 if (!function.evaluate(x, f) || !function.first_derivative(x, J)) {
913 return false;
914 }
915 least_squares_gradient(f, J, out);
916 return true;
917 }
918 };
919
920 auto first_derivative_lambda = [&function](const Input &x,
921 FirstDerivative &out) -> bool {
922 if constexpr (Kind == Problem::ROOT || Kind == Problem::COST) {
923 return function.first_derivative(x, out);
924 } else if constexpr (Kind == Problem::LEAST_SQUARES) {
925 FunctionOutput f;
926 FunctionFirstDerivative J;
927 if (!function.evaluate(x, f) || !function.first_derivative(x, J)) {
928 return false;
929 }
930 least_squares_gradient(f, J, out);
931 return true;
932 } else if constexpr (Kind == Problem::STATIONARY_COST) {
933 return function.second_derivative(x, out);
934 } else {
935 FunctionOutput f;
936 FunctionFirstDerivative J;
937 FunctionSecondDerivative H;
938 if (!function.evaluate(x, f) || !function.first_derivative(x, J) ||
939 !function.second_derivative(x, H)) {
940 return false;
941 }
942 least_squares_hessian(f, J, H, out);
943 return true;
944 }
945 };
946
947 auto second_derivative_lambda = [&function](
948 const Input &x,
949 SecondDerivative &out) -> bool {
950 if constexpr (Kind == Problem::ROOT || Kind == Problem::COST) {
951 return function.second_derivative(x, out);
952 } else if constexpr (Kind == Problem::LEAST_SQUARES) {
953 FunctionOutput f;
954 FunctionFirstDerivative J;
955 FunctionSecondDerivative H;
956 if (!function.evaluate(x, f) || !function.first_derivative(x, J) ||
957 !function.second_derivative(x, H)) {
958 return false;
959 }
960 least_squares_hessian(f, J, H, out);
961 return true;
962 } else {
963 static_cast<void>(function);
964 static_cast<void>(x);
965 static_cast<void>(out);
966 OPTIMIST_ERROR(CMD "second derivative not available for "
967 "stationary-point problems.");
968 return false;
969 }
970 };
971
972 // Select solver method based on derivative requirements
973 if constexpr (DerivedSolver::RequiresFunction &&
974 !DerivedSolver::RequiresFirstDerivative &&
975 !DerivedSolver::RequiresSecondDerivative) {
976 return static_cast<DerivedSolver *>(this)->solve_impl(function_lambda,
977 x_ini,
978 x_sol);
979 } else if constexpr (DerivedSolver::RequiresFunction &&
980 DerivedSolver::RequiresFirstDerivative &&
981 !DerivedSolver::RequiresSecondDerivative) {
982 return static_cast<DerivedSolver *>(this)->solve_impl(
983 function_lambda,
984 first_derivative_lambda,
985 x_ini,
986 x_sol);
987 } else if constexpr (DerivedSolver::RequiresFunction &&
988 DerivedSolver::RequiresFirstDerivative &&
989 DerivedSolver::RequiresSecondDerivative) {
990 return static_cast<DerivedSolver *>(this)->solve_impl(
991 function_lambda,
992 first_derivative_lambda,
993 second_derivative_lambda,
994 x_ini,
995 x_sol);
996 } else {
997 static_assert(DerivedSolver::RequiresFunction,
998 CMD "no matching function signature for the solver.");
999 return false;
1000 }
1001
1002#undef CMD
1003 }

1094 template <typename FunctionLambda>
1095 bool damp(FunctionLambda &&function,
1096 const Input &x_old,
1097 const Input &function_old,
1098 const Input &step_old,
1099 Input &x_new,
1100 Input &function_new,
1101 Input &step_new) {
1102#define CMD "Optimist::Solver::damp(...): "
1103
1104 Scalar step_norm_old, step_norm_new, residuals_old, residuals_new,
1105 tau{1.0};
1106 for (this->m_relaxations = 0;
1107 this->m_relaxations < this->m_max_relaxations;
1108 ++this->m_relaxations) {
1109 // Update point
1110 step_new = tau * step_old;
1111 x_new = x_old + step_new;
1112 bool success{
1113 this->evaluate_function(function,
1114 x_new,
1115 function_new)};
1116 OPTIMIST_ASSERT(success,
1117 CMD "function evaluation failed during damping.");
1118
1119 // Compute step norms
1120 if constexpr (InputTrait::IsEigen) {
1121 step_norm_old = step_old.norm();
1122 step_norm_new = step_new.norm();
1123 } else {
1124 step_norm_old = std::abs(step_old);
1125 step_norm_new = std::abs(step_new);
1126 }
1127
1128 // Compute residual norms
1129 if constexpr (OutputTrait::IsEigen) {
1130 residuals_old = function_old.norm();
1131 residuals_new = function_new.norm();
1132 } else {
1133 residuals_old = std::abs(function_old);
1134 residuals_new = std::abs(function_new);
1135 }
1136
1137 // Check relaxation
1138 if (residuals_new < residuals_old ||
1139 step_norm_new < (Scalar(1.0) - tau / Scalar(2.0)) * step_norm_old) {
1140 return true;
1141 } else {
1142 tau *= this->m_alpha;
1143 }
1144 }
1145 return false;
1146
1147#undef CMD
1148 }
