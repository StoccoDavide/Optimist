138 template <typename FunctionLambda, typename GradientLambda>
139 bool solve_impl(FunctionLambda &&function,
140 GradientLambda &&gradient,
141 const Vector &x_ini,
142 Vector &x_sol) {
143#define CMD "Optimist::Optimizer::BFGS::solve_impl(...): "
144
145 this->reset_counters();
146 if (this->m_verbose) {
147 this->header();
148 }
149
150 const Integer n{static_cast<Integer>(x_ini.size())};
151 Dense x(Utils::to_dense(x_ini)), g(n), d(n), s(n), y(n);
152 Scalar f;
153 DenseMatrix H(DenseMatrix::Identity(n, n));
154 std::deque<std::pair<Dense, Dense>> pairs; // L-BFGS (s, y) pairs
155 Scalar gamma{1.0}; // L-BFGS scaling
156
157 // Evaluate the objective and the gradient at a dense point
158 auto eval_f = [this, &function](const Dense &z, Scalar &out) -> bool {
159 return this->evaluate_function(function,
160 Utils::from_dense<Vector>(z),
161 out) &&
162 std::isfinite(out);
163 };
164 auto eval_g = [this, &gradient](const Dense &z, Dense &out) -> bool {
165 FirstDerivative tmp;
166 bool ok{this->evaluate_gradient(gradient,
167 Utils::from_dense<Vector>(z),
168 tmp)};
169 if (!ok) {
170 return false;
171 }
172 out = Utils::to_dense(tmp);
173 return out.allFinite();
174 };
175
176 OPTIMIST_ASSERT(eval_f(x, f) && eval_g(x, g),
177 CMD "evaluation failed at the initial point.");
178
179 std::string notes{"Initialize"};
180 bool fresh{true}; // Current direction is steepest descent
181 for (this->m_iterations = 1;
182 this->m_iterations < this->m_max_iterations;
183 ++this->m_iterations) {
184 const Scalar g_norm{g.norm()};
185 if (this->m_verbose) {
186 this->info(g_norm, notes);
187 }
188 if (g_norm < this->m_tolerance) {
189 this->m_converged = true;
190 break;
191 }
192
193 // Search direction
194 if (this->m_method == Method::BFGS) {
195 d.noalias() = -H * g;
196 } else {
197 d = -this->two_loop(pairs, gamma, g);
198 }
199 if (!(g.dot(d) < -EPSILON * g_norm * d.norm())) {
200 d = -g;
201 fresh = true;
202 notes = "Restart";
203 H.setIdentity();
204 pairs.clear();
205 }
206
207 // Line search along d
208 Dense x_trial(n), g_trial(n);
209 Scalar f_trial{f};
210 bool g_valid{false};
211 auto phi = [&](const Scalar a, Scalar &v, Scalar *dv) -> bool {
212 x_trial = x + a * d;
213 g_valid = false;
214 if (!eval_f(x_trial, v)) {
215 return false;
216 }
217 f_trial = v;
218 if (dv != nullptr) {
219 if (!eval_g(x_trial, g_trial)) {
220 return false;
221 }
222 g_valid = true;
223 *dv = g_trial.dot(d);
224 }
225 return true;
226 };
227 Scalar alpha{fresh ? std::min(Scalar(1), 1 / g_norm) : Scalar(1)};
228 Scalar phi_alpha, dphi_alpha;
229 bool accepted{false};
230 try {
231 accepted = LineSearch::search(this->m_line_search, phi, f,
232 g.dot(d), alpha, phi_alpha,
233 dphi_alpha, this->m_settings);
234 } catch (const std::runtime_error &) {
235 notes = "Budget exhausted";
236 break;
237 }
238 if (!accepted) {
239 if (fresh) {
240 notes = "Line search failed";
241 break;
242 }
243 // Retry from steepest descent with a fresh approximation
244 H.setIdentity();
245 pairs.clear();
246 fresh = true;
247 notes = "Restart";
248 continue;
249 }
250 // Make sure the stored trial point is the accepted one
251 if ((x_trial - (x + alpha * d)).norm() > 0) {
252 x_trial = x + alpha * d;
253 OPTIMIST_ASSERT(eval_f(x_trial, f_trial),
254 CMD "function evaluation failed.");
255 g_valid = false;
256 }
257 if (!g_valid) {
258 OPTIMIST_ASSERT(eval_g(x_trial, g_trial),
259 CMD "gradient evaluation failed.");
260 }
261
262 s = x_trial - x;
263 y = g_trial - g;
264 x = x_trial;
265 f = f_trial;
266 g = g_trial;
267
268 // Stop on stagnation
269 if (s.norm() <= EPSILON * (1 + x.norm())) {
270 this->m_converged = g.norm() < this->m_tolerance;
271 notes = "Stagnation";
272 break;
273 }
274
275 // Inverse Hessian update (skipped if the curvature condition fails)
276 const Scalar sy{s.dot(y)};
277 if (sy > SQRT_EPSILON * s.norm() * y.norm()) {
278 const Scalar rho{1 / sy};
279 if (this->m_method == Method::BFGS) {
280 if (fresh) {
281 // Shanno-Phua scaling of the initial approximation
282 H.setIdentity();
283 H *= sy / y.squaredNorm();
284 }
285 const Dense Hy{H * y};
286 const Scalar yHy{y.dot(Hy)};
287 H.noalias() -= rho * (s * Hy.transpose() + Hy * s.transpose());
288 H.noalias() += (rho * rho * yHy + rho) * (s * s.transpose());
289 } else {
290 pairs.emplace_back(s, y);
291 if (static_cast<Integer>(pairs.size()) > this->m_memory) {
292 pairs.pop_front();
293 }
294 gamma = sy / y.squaredNorm();
295 }
296 notes = "Update";
297 fresh = false;
298 } else {
299 notes = "Skip update";
300 }
301 }
302
303 if (this->m_verbose) {
304 this->bottom();
305 }
306 x_sol = Utils::from_dense<Vector>(x);
307 return this->m_converged;
308
309#undef CMD
310 }
311
312 private:
316 static Dense two_loop(const std::deque<std::pair<Dense, Dense>> &pairs,
317 const Scalar gamma,
318 const Dense &g) {
319 Dense q(g);
320 std::vector<Scalar> a(pairs.size());
321 for (Integer i{static_cast<Integer>(pairs.size()) - 1}; i >= 0; --i) {
322 const auto &[s, y] = pairs[i];
323 a[i] = s.dot(q) / y.dot(s);
324 q -= a[i] * y;
325 }
326 Dense r(gamma * q);
327 for (Integer i{0}; i < static_cast<Integer>(pairs.size()); ++i) {
328 const auto &[s, y] = pairs[i];
329 const Scalar b{y.dot(r) / y.dot(s)};
330 r += (a[i] - b) * s;
331 }
332 return r;
333 }
