102 template <typename FunctionLambda>
103 bool solve_impl(FunctionLambda &&function,
104 const Vector &x_ini,
105 Vector &x_sol) {
106 this->reset_counters();
107 if (this->m_verbose) {
108 this->header();
109 }
110
111 const Integer n{static_cast<Integer>(x_ini.size())};
112 Dense base(Utils::to_dense(x_ini));
113 Dense sign(Dense::Ones(n)); // Preferred direction per coordinate
114 Scalar h{this->m_step};
115 Scalar f_base{this->eval(function, base)};
116 std::string notes{"Initialize"};
117
118 try {
119 for (this->m_iterations = 1;
120 this->m_iterations < this->m_max_iterations;
121 ++this->m_iterations) {
122 if (this->m_verbose) {
123 this->info(h, notes);
124 }
125 if (h < this->m_tolerance * (1 + base.template lpNorm<Eigen::Infinity>())) {
126 this->m_converged = true;
127 break;
128 }
129
130 // Exploratory moves around the base point
131 Dense x(base);
132 Scalar f{f_base};
133 this->explore(function, x, f, sign, h);
134 if (!(f < f_base)) {
135 h *= this->m_rho;
136 notes = "Shrink";
137 continue;
138 }
139
140 // Pattern moves: keep jumping along the successful direction as
141 // long as the exploration around the jump improves
142 notes = "Explore";
143 while (true) {
144 Dense jump{2 * x - base};
145 base = x;
146 f_base = f;
147 Scalar f_jump{this->eval(function, jump)};
148 this->explore(function, jump, f_jump, sign, h);
149 if (f_jump < f_base) {
150 x = jump;
151 f = f_jump;
152 notes = "Pattern";
153 } else {
154 break;
155 }
156 }
157 }
158 } catch (const std::runtime_error &) {
159 notes = "Budget exhausted";
160 }
161
162 if (this->m_verbose) {
163 this->bottom();
164 }
165 x_sol = Utils::from_dense<Vector>(base);
166 return this->m_converged;
167 }

184 template <typename FunctionLambda>
185 void explore(FunctionLambda &function,
186 Dense &x,
187 Scalar &f,
188 Dense &sign,
189 const Scalar h) {
190 for (Integer j{0}; j < static_cast<Integer>(x.size()); ++j) {
191 const Scalar x_j{x(j)};
192 x(j) = x_j + sign(j) * h;
193 Scalar f_try{this->eval(function, x)};
194 if (f_try < f) {
195 f = f_try;
196 continue;
197 }
198 x(j) = x_j - sign(j) * h;
199 f_try = this->eval(function, x);
200 if (f_try < f) {
201 f = f_try;
202 sign(j) = -sign(j);
203 continue;
204 }
205 x(j) = x_j;
206 }
207 }
