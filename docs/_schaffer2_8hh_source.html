68 // f = 1/2 + A * B, with A = sin^2(u) - 1/2, u = x^2 - y^2, and
69 // B = D^(-2), D = 1 + (x^2 + y^2)/1000
70
71 Scalar value(const Dense &x) const {
72 const Scalar u{x(0) * x(0) - x(1) * x(1)};
73 const Scalar D{1 + (x(0) * x(0) + x(1) * x(1)) / 1000};
74 const Scalar s{std::sin(u)};
75 return Scalar(0.5) + (s * s - Scalar(0.5)) / (D * D);
76 }

78 void gradient(const Dense &x, Dense &g) const {
79 Scalar A, B;
80 Dense dA(2), dB(2);
81 DenseMatrix ddA(2, 2), ddB(2, 2);
82 this->terms(x, A, B, dA, dB, ddA, ddB);

84 }

90 this->terms(x, A, B, dA, dB, ddA, ddB);

95 void terms(const Dense &x,
96 Scalar &A,
97 Scalar &B,
98 Dense &dA,
99 Dense &dB,
100 DenseMatrix &ddA,
101 DenseMatrix &ddB) const {
102 const Scalar u{x(0) * x(0) - x(1) * x(1)};
103 const Scalar D{1 + (x(0) * x(0) + x(1) * x(1)) / 1000};
104 const Scalar s{std::sin(u)}, s2{std::sin(2 * u)}, c2{std::cos(2 * u)};
105 A = s * s - Scalar(0.5);
106 B = 1 / (D * D);
107 // dA = sin(2u) * du, du = (2x, -2y)
108 dA(0) = 2 * x(0) * s2;
109 dA(1) = -2 * x(1) * s2;
110 ddA(0, 0) = 2 * s2 + 8 * x(0) * x(0) * c2;
111 ddA(1, 1) = -2 * s2 + 8 * x(1) * x(1) * c2;
112 ddA(0, 1) = -8 * x(0) * x(1) * c2;
113 ddA(1, 0) = ddA(0, 1);
114 // dB = -2 D^(-3) dD, dD = (x, y)/500
115 const Scalar D3{D * D * D}, D4{D3 * D};
116 dB(0) = -x(0) / (250 * D3);
117 dB(1) = -x(1) / (250 * D3);
118 ddB(0, 0) = -1 / (250 * D3) + 3 * x(0) * x(0) / (125000 * D4);
119 ddB(1, 1) = -1 / (250 * D3) + 3 * x(1) * x(1) / (125000 * D4);
120 ddB(0, 1) = 3 * x(0) * x(1) / (125000 * D4);
121 ddB(1, 0) = ddB(0, 1);
122 }
