113 template <typename FunctionLambda, typename JacobianLambda>
114 bool solve_impl(FunctionLambda &&function,
115 JacobianLambda &&jacobian,
116 const Vector &x_ini,
117 Vector &x_sol) {
118#define CMD "Optimist::RootFinder::LevenbergMarquardt::solve_impl(...): "
119
120 this->reset_counters();
121 if (this->m_verbose) {
122 this->header();
123 }
124
125 const Integer n{static_cast<Integer>(x_ini.size())};
126 Dense x(Utils::to_dense(x_ini)), F, g(n), h(n), D(n);
127 DenseMatrix J, A(n, n);
128
129 auto eval_F = [this, &function](const Dense &z, Dense &out) -> bool {
130 OutVector tmp;
131 bool ok{this->evaluate_function(function,
132 Utils::from_dense<Vector>(z),
133 tmp)};
134 if (!ok) {
135 return false;
136 }
137 out = Utils::to_dense(tmp);
138 return out.allFinite();
139 };
140 auto eval_J = [this, &jacobian](const Dense &z,
141 DenseMatrix &out) -> bool {
142 FirstDerivative tmp;
143 bool ok{this->evaluate_jacobian(jacobian,
144 Utils::from_dense<Vector>(z),
145 tmp)};
146 if (!ok) {
147 return false;
148 }
149 out = DenseMatrix(tmp);
150 return out.allFinite();
151 };
152
153 OPTIMIST_ASSERT(eval_F(x, F) && eval_J(x, J),
154 CMD "evaluation failed at the initial point.");
155 A = J.transpose() * J;
156 g = J.transpose() * F;
157 Scalar lambda{this->m_tau * A.diagonal().maxCoeff()};
158 if (!(lambda > 0)) {
159 lambda = this->m_tau;
160 }
161 Scalar nu{2};
162
163 std::string notes{"Initialize"};
164 for (this->m_iterations = 1;
165 this->m_iterations < this->m_max_iterations;
166 ++this->m_iterations) {
167 const Scalar residuals{F.norm()};
168 if (this->m_verbose) {
169 this->info(residuals, notes);
170 }
171 if (residuals < this->m_tolerance) {
172 this->m_converged = true;
173 break;
174 }
175 // Overdetermined systems may have no root: stop at a stationary
176 // point of |F|^2/2, i.e., where J'F vanishes relative to |F|
177 if (F.size() > n &&
178 g.norm() < this->m_tolerance * std::max(Scalar(1), residuals)) {
179 this->m_converged = true;
180 notes = "Least-squares solution";
181 break;
182 }
183
184 // Damped normal equations: (J'J + lambda D) h = -J'F
185 if (this->m_scaling) {
186 D = A.diagonal().cwiseMax(SQRT_EPSILON);
187 } else {
188 D.setOnes();
189 }
190 DenseMatrix M{A};
191 M.diagonal() += lambda * D;
192 const Eigen::LDLT<DenseMatrix> ldlt(M);
193 h = -ldlt.solve(g);
194 if (ldlt.info() != Eigen::Success || !h.allFinite()) {
195 lambda *= nu;
196 nu *= 2;
197 notes = "Singular system";
198 continue;
199 }
200 if (h.norm() <= EPSILON * (x.norm() + EPSILON)) {
201 notes = "Stagnation";
202 break;
203 }
204
205 // Gain ratio between actual and predicted reduction of |F|^2/2
206 Dense x_new{x + h}, F_new;
207 bool ok{false};
208 try {
209 ok = eval_F(x_new, F_new);
210 } catch (const std::runtime_error &) {
211 notes = "Budget exhausted";
212 break;
213 }
214 const Scalar predicted{Scalar(0.5) * h.dot(lambda * D.cwiseProduct(h) - g)};
215 const Scalar actual{ok ? Scalar(0.5) * (F.squaredNorm() - F_new.squaredNorm())
216 : -INFTY};
217 const Scalar rho{predicted > 0 ? actual / predicted : Scalar(-1)};
218
219 if (rho > 0) {
220 // Accept the step and relax the damping (Nielsen's update)
221 x = x_new;
222 F = F_new;
223 try {
224 OPTIMIST_ASSERT(eval_J(x, J), CMD "jacobian evaluation failed.");
225 } catch (const std::runtime_error &) {
226 notes = "Budget exhausted";
227 break;
228 }
229 A = J.transpose() * J;
230 g = J.transpose() * F;
231 const Scalar t{2 * rho - 1};
232 lambda *= std::max(Scalar(1) / 3, 1 - t * t * t);
233 nu = 2;
234 notes = "Accept";
235 } else {
236 lambda *= nu;
237 nu *= 2;
238 notes = "Reject";
239 }
240 }
241
242 if (this->m_verbose) {
243 this->bottom();
244 }
245 x_sol = Utils::from_dense<Vector>(x);
246 return this->m_converged;
247
248#undef CMD
249 }
