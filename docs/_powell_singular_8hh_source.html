60 void residual(const Dense &x, Dense &F) const {
61 const Scalar s5{std::sqrt(Scalar(5))}, s10{std::sqrt(Scalar(10))};
62 F(0) = x(0) + 10 * x(1);
63 F(1) = s5 * (x(2) - x(3));
64 F(2) = (x(1) - 2 * x(2)) * (x(1) - 2 * x(2));
65 F(3) = s10 * (x(0) - x(3)) * (x(0) - x(3));
66 }

68 void jacobian(const Dense &x, DenseMatrix &J) const {
69 const Scalar s5{std::sqrt(Scalar(5))}, s10{std::sqrt(Scalar(10))};
70 const Scalar a{x(1) - 2 * x(2)}, b{x(0) - x(3)};
71 J.setZero();
72 J(0, 0) = 1;
73 J(0, 1) = 10;
74 J(1, 2) = s5;
75 J(1, 3) = -s5;
76 J(2, 1) = 2 * a;
77 J(2, 2) = -4 * a;
78 J(3, 0) = 2 * s10 * b;
79 J(3, 3) = -2 * s10 * b;
