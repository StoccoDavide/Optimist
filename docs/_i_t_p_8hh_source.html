84 template <typename FunctionLambda>
85 Scalar find_root_impl(FunctionLambda &&function) {
86#define CMD "Optimist::RootFinder::ITP::find_root_impl(...): "
87
88 Scalar a{this->m_a}, b{this->m_b}, fa{this->m_fa}, fb{this->m_fb};
89 if (a > b) {
90 std::swap(a, b);
91 std::swap(fa, fb);
92 }
93 if (fa == 0 || fb == 0) {
94 this->m_converged = true;
95 return fa == 0 ? a : b;
96 }
97
98 // Orient the function so that f(a) < 0 < f(b)
99 const Scalar sign{fa < 0 ? Scalar(1) : Scalar(-1)};
100 fa *= sign;
101 fb *= sign;
102
103 const Scalar eps{std::max(
104 this->m_tolerance_bracketing,
105 EPSILON * std::max({Scalar(1), std::abs(a), std::abs(b)}))};
106 const Scalar kappa_1{this->m_kappa_1 / (b - a)};
107 const Integer n_half{static_cast<Integer>(
108 std::ceil(std::log2((b - a) / (2 * eps))))};
109 const Integer n_max{std::max(n_half, Integer(0)) + this->m_n_0};
110
111 for (Integer j{0}; this->m_iterations < this->m_max_iterations; ++j) {
112 ++this->m_iterations;
113 if (this->m_verbose) {
114 this->info(std::min(std::abs(fa), std::abs(fb)));
115 }
116 if (b - a <= 2 * eps) {
117 this->m_converged = true;
118 break;
119 }
120
121 // Interpolation (regula falsi)
122 const Scalar x_half{(a + b) / 2};
123 const Scalar x_f{(fb * a - fa * b) / (fb - fa)};
124
125 // Truncation
126 const Scalar sigma{x_half >= x_f ? Scalar(1) : Scalar(-1)};
127 const Scalar delta{kappa_1 * std::pow(b - a, this->m_kappa_2)};
128 const Scalar x_t{delta <= std::abs(x_half - x_f) ? x_f + sigma * delta
129 : x_half};
130
131 // Projection onto the minmax interval
132 const Scalar r{eps * std::ldexp(Scalar(1), n_max - j) - (b - a) / 2};
133 const Scalar x_itp{std::abs(x_t - x_half) <= r ? x_t
134 : x_half - sigma * r};
135
136 // Update the bracket
137 Scalar f_itp;
138 bool success{this->evaluate_function(function, x_itp, f_itp)};
139 OPTIMIST_ASSERT(success && std::isfinite(f_itp),
140 CMD "function evaluation failed at iteration "
141 << this->m_iterations << ".");
142 f_itp *= sign;
143 if (f_itp > 0) {
144 b = x_itp;
145 fb = f_itp;
146 } else if (f_itp < 0) {
147 a = x_itp;
148 fa = f_itp;
149 } else {
150 a = b = x_itp;
151 fa = fb = 0;
152 this->m_converged = true;
153 break;
154 }
155 }
156
157 this->m_a = a;
158 this->m_b = b;
159 this->m_fa = sign * fa;
160 this->m_fb = sign * fb;
161 return std::abs(fa) < std::abs(fb) ? a : b;
162
163#undef CMD
164 }
