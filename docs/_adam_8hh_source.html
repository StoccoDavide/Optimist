127 template <typename FunctionLambda, typename GradientLambda>
128 bool solve_impl(FunctionLambda &&function,
129 GradientLambda &&gradient,
130 const Vector &x_ini,
131 Vector &x_sol) {
132#define CMD "Optimist::Optimizer::Adam::solve_impl(...): "
133
134 this->reset_counters();
135 if (this->m_verbose) {
136 this->header();
137 }
138
139 const Integer n{static_cast<Integer>(x_ini.size())};
140 Dense x(Utils::to_dense(x_ini)), g(n), m(Dense::Zero(n)),
141 v(Dense::Zero(n)), v_hat_max(Dense::Zero(n));
142 Dense x_best(x);
143 Scalar f, f_best{INFTY};
144
145 auto evaluate = [&](const Dense &z) -> bool {
146 const Vector zv{Utils::from_dense<Vector>(z)};
147 FirstDerivative gv;
148 if (!this->evaluate_function(function, zv, f) ||
149 !this->evaluate_gradient(gradient, zv, gv)) {
150 return false;
151 }
152 g = Utils::to_dense(gv);
153 return std::isfinite(f) && g.allFinite();
154 };
155
156 Scalar beta_1_t{1}, beta_2_t{1};
157 std::string notes{"Initialize"};
158 try {
159 OPTIMIST_ASSERT(evaluate(x), CMD "evaluation failed.");
160 for (this->m_iterations = 1;
161 this->m_iterations < this->m_max_iterations;
162 ++this->m_iterations) {
163 if (f < f_best) {
164 f_best = f;
165 x_best = x;
166 }
167 const Scalar g_norm{g.norm()};
168 if (this->m_verbose && (this->m_iterations % 100 == 1)) {
169 this->info(g_norm, notes);
170 }
171 if (g_norm < this->m_tolerance) {
172 this->m_converged = true;
173 x_best = x;
174 break;
175 }
176
177 // Biased moment estimates and bias corrections
178 m = this->m_beta_1 * m + (1 - this->m_beta_1) * g;
179 v = this->m_beta_2 * v + (1 - this->m_beta_2) * g.cwiseAbs2();
180 beta_1_t *= this->m_beta_1;
181 beta_2_t *= this->m_beta_2;
182 const Dense m_hat{m / (1 - beta_1_t)};
183 Dense v_hat{v / (1 - beta_2_t)};
184 if (this->m_amsgrad) {
185 v_hat_max = v_hat_max.cwiseMax(v_hat);
186 v_hat = v_hat_max;
187 }
188
189 const Scalar eta{this->m_learning_rate /
190 (1 + this->m_decay * this->m_iterations)};
191 x -= eta * m_hat.cwiseQuotient(
192 (v_hat.cwiseSqrt().array() + this->m_epsilon).matrix());
193 OPTIMIST_ASSERT(evaluate(x), CMD "evaluation failed.");
194 notes = "-";
195 }
196 } catch (const std::runtime_error &) {
197 notes = "Budget exhausted";
198 }
199
200 if (this->m_verbose) {
201 this->bottom();
202 }
203 x_sol = Utils::from_dense<Vector>(x_best);
204 return this->m_converged;
205
206#undef CMD
207 }
