82 template <typename FunctionLambda>
83 bool bracketing(FunctionLambda &&function) {
84#define CMD "Optimist::RootFinder::Algo748::bracketing(...): "
85
86 {
87 Scalar tolerance{static_cast<Scalar>(0.7) *
88 this->m_tolerance_bracketing};
89 Scalar hba{(this->m_b - this->m_a) / static_cast<Scalar>(2.0)};
90 if (hba <= tolerance) {
91 this->m_c = this->m_a + hba;
92 } else if (this->m_c <= this->m_a + tolerance) {
93 this->m_c = this->m_a + tolerance;
94 } else if (this->m_c >= this->m_b - tolerance) {
95 this->m_c = this->m_b - tolerance;
96 }
97 }
98
99 // If f(c) = 0 set a = c and return true
100 bool success{
101 this->evaluate_function(function,
102 this->m_c,
103 this->m_fc)};
104 OPTIMIST_ASSERT(success,
105 CMD "function evaluation failed during bracketing.");
106 if (this->m_fc == 0) {
107 this->m_a = this->m_c;
108 this->m_fa = 0.0;
109 this->m_d = 0.0;
110 this->m_fd = 0.0;
111 return true;
112 } else {
113 // If f(c) is not zero, then determine the new enclosing interval
114 if (this->m_fa * this->m_fc < 0) {
115 // D <- B <- C
116 this->m_d = this->m_b;
117 this->m_fd = this->m_fb;
118 this->m_b = this->m_c;
119 this->m_fb = this->m_fc;
120 } else {
121 // D <- A <- C
122 this->m_d = this->m_a;
123 this->m_fd = this->m_fa;
124 this->m_a = this->m_c;
125 this->m_fa = this->m_fc;
126 }
127 return false;
128 }
129
130#undef CMD
131 }

140 Scalar cubic_interpolation() {
141 // Compute the coefficients for the inverse cubic interpolation
142 Scalar d_1{this->m_b - this->m_a};
143 Scalar d_2{this->m_d - this->m_a};
144 Scalar d_3{this->m_e - this->m_a};
145
146 Scalar dd_0{d_1 / (this->m_fb - this->m_fa)};
147 Scalar dd_1{(d_1 - d_2) / (this->m_fb - this->m_fd)};
148 Scalar dd_2{(d_2 - d_3) / (this->m_fd - this->m_fe)};
149
150 Scalar ddd_0{(dd_0 - dd_1) / (this->m_fa - this->m_fd)};
151 Scalar ddd_1{(dd_1 - dd_2) / (this->m_fb - this->m_fe)};
152
153 Scalar dddd_0{(ddd_0 - ddd_1) / (this->m_fa - this->m_fe)};
154
155 // Compute the approximate root
156 Scalar root{this->m_a -
157 this->m_fa *
158 (dd_0 - this->m_fb * (ddd_0 - this->m_fd * dddd_0))};
159 Scalar tolerance{static_cast<Scalar>(0.7) *
160 this->m_tolerance_bracketing};
161 if (root <= this->m_a + tolerance || root >= this->m_b - tolerance) {
162 root = (this->m_a + this->m_b) / 2.0;
163 }
164 return root;
165 }

175 Scalar newton_quadratic(Integer n) {
176 // Compute the coefficients for the quadratic interpolation
177 Scalar a_0{this->m_fa};
178 Scalar a_1{(this->m_fb - this->m_fa) / (this->m_b - this->m_a)};
179 Scalar a_2{((this->m_fd - this->m_fb) / (this->m_d - this->m_b) - a_1) /
180 (this->m_d - this->m_a)};
181
182 // Safeguard to avoid overflow
183 if (a_2 == 0) {
184 return this->m_a - a_0 / a_1;
185 }
186
187 // Determine the starting point of newton steps
188 Scalar c{a_2 * this->m_fa > 0 ? this->m_a : this->m_b};
189
190 // Start the safeguarded newton steps
191 bool is_nonzero{true};
192 Scalar pc{0.0}, pdc{0.0};
193 for (Integer i{0}; i < n && is_nonzero; ++i) {
194 pc = a_0 + (a_1 + a_2 * (c - this->m_b)) * (c - this->m_a);
195 pdc = a_1 + a_2 * ((2.0 * c) - (this->m_a + this->m_b));
196 is_nonzero = pdc != 0;
197 if (is_nonzero) {
198 c -= pc / pdc;
199 }
200 }
201
202 if (is_nonzero) {
203 return c;
204 } else {
205 return this->m_a - a_0 / a_1;
206 }

228 template <typename FunctionLambda>
229 Scalar find_root_impl(FunctionLambda &&function) {
230#define CMD "Optimist::RootFinder::Algo748::find_root_impl(...): "
231
232 bool success;
233 Scalar tolerance_step{this->m_tolerance_bracketing};
234 Scalar tolerance_function{this->m_tolerance_bracketing};
235
236 // Check for trivial solution
237 this->m_converged = this->m_fa == 0;
238 if (this->m_verbose) {
239 this->info(std::abs(this->m_fa));
240 }
241 if (this->m_converged) {
242 return this->m_fa;
243 }
244 this->m_converged = this->m_fb == 0;
245 if (this->m_verbose) {
246 this->info(std::abs(this->m_fb));
247 }
248 if (this->m_converged) {
249 return this->m_fb;
250 }
251
252 // Initialize to dumb values
253 this->m_e = QUIET_NAN;
254 this->m_fe = QUIET_NAN;
255
256 // While f(left) or f(right) are infinite perform bisection
257 while (!(std::isfinite(this->m_fa) && std::isfinite(this->m_fb))) {
258 ++this->m_iterations;
259 this->m_c = (this->m_a + this->m_b) / 2.0;
260 success =
261 this->evaluate_function(function,
262 this->m_c,
263 this->m_fc);
264 OPTIMIST_ASSERT(success,
265 CMD "function evaluation failed at iteration "
266 << this->m_iterations << ".");
267 this->m_converged = this->m_fc == 0;
268 if (this->m_verbose) {
269 this->info(std::abs(this->m_fc));
270 }
271 if (this->m_converged) {
272 return this->m_fc;
273 }
274 if (this->m_fa * this->m_fc < 0) {
275 // -> [a, c]
276 this->m_b = this->m_c;
277 this->m_fb = this->m_fc;
278 } else {
279 // -> [c, b]
280 this->m_a = this->m_c;
281 this->m_fa = this->m_fc;
282 }
283
284 // Check for convergence
285 Scalar abs_fa{std::abs(this->m_fa)};
286 Scalar abs_fb{std::abs(this->m_fb)};
287 if (this->m_verbose) {
288 this->info(abs_fa < abs_fb ? abs_fa : abs_fb);
289 }
290 this->m_converged = (this->m_b - this->m_a) < tolerance_step ||
291 abs_fa < tolerance_function ||
292 abs_fb < tolerance_function;
293 if (this->m_converged) {
294 return abs_fb < abs_fa ? this->m_b : this->m_a;
295 }
296 }
297
298 {
299 // Impede the step c to be too close to the bounds a or b.
300 Scalar b_a{this->m_b - this->m_a};
301 Scalar r{b_a / (this->m_fb - this->m_fa)};
302 this->m_c = std::abs(this->m_fb) < std::abs(this->m_fa)
303 ? this->m_b + this->m_fb * r
304 : this->m_a - this->m_fa * r;
305 Scalar delta{this->m_interval_shink * b_a}, a_min, b_max;
306 if (this->m_c < (a_min = this->m_a + delta)) {
307 this->m_c = a_min;
308 } else if (this->m_c > (b_max = this->m_b - delta)) {
309 this->m_c = b_max;
310 }
311 }
312
313 // Call "bracketing" to get a shrinked enclosing interval and to update
314 // the termination criterion
315 this->m_converged =
316 this->bracketing(function);
317 if (this->m_verbose) {
318 this->info(std::abs(this->m_fa));
319 }
320 if (this->m_converged) {
321 return this->m_a;
322 }
323 this->m_converged = false;
324
325 // The enclosing interval is recorded as [a0, b0] before executing the
326 // iteration steps
327 while (++this->m_iterations < this->m_max_iterations) {
328 ++this->m_iterations;
329 Scalar a0_b0{this->m_b - this->m_a};
330
331 // Check the termination criterion
332 {
333 Scalar abs_fa{std::abs(this->m_fa)};
334 Scalar abs_fb{std::abs(this->m_fb)};
335 if (this->m_verbose) {
336 this->info(abs_fa < abs_fb ? abs_fa : abs_fb);
337 }
338 this->m_converged =
339 abs_fa < tolerance_function || abs_fb < tolerance_function;
340 if (this->m_converged) {
341 return abs_fa < abs_fb ? this->m_a : this->m_b;
342 }
343 }
344
345 // Starting with the second iteration: newton quadratic or cubic
346 // interpolation
347 bool do_newton_quadratic{false};
348 if (!std::isfinite(this->m_fe)) {
349 do_newton_quadratic = true;
350 } else if (!this->all_different(this->m_fa,
351 this->m_fb,
352 this->m_fd,
353 this->m_fe)) {
354 do_newton_quadratic = true;
355 } else {
356 this->m_c = this->cubic_interpolation();
357 do_newton_quadratic =
358 (this->m_c - this->m_a) * (this->m_c - this->m_b) >= 0.0;
359 }
360 if (do_newton_quadratic) {
361 this->m_c = this->newton_quadratic(2);
362 }
363
364 this->m_e = this->m_d;
365 this->m_fe = this->m_fd;
366
367 // Call "bracketing" to get a shrinked enclosing interval and to
368 // update the termination criterion
369 this->m_converged =
370 this->bracketing(function) ||
371 (this->m_b - this->m_a) <= this->m_tolerance;
372 if (this->m_verbose) {
373 this->info(std::abs(this->m_fa) < std::abs(this->m_fb)
374 ? std::abs(this->m_fa)
375 : std::abs(this->m_fb));
376 }
377 if (this->m_converged) {
378 return std::abs(this->m_fa) < std::abs(this->m_fb) ? this->m_a
379 : this->m_b;
380 }
381 if (!this->all_different(this->m_fa,
382 this->m_fb,
383 this->m_fd,
384 this->m_fe)) {
385 do_newton_quadratic = true;
386 } else {
387 this->m_c = this->cubic_interpolation();
388 do_newton_quadratic =
389 (this->m_c - this->m_a) * (this->m_c - this->m_b) >= 0.0;
390 }
391 if (do_newton_quadratic) {
392 this->m_c = this->newton_quadratic(3);
393 }
394
395 // Call "bracketing" to get a shrinked enclosing interval and to
396 // update the termination criterion
397 {
398 this->m_converged =
399 this->bracketing(function) ||
400 (this->m_b - this->m_a) <= this->m_tolerance;
401 Scalar abs_fa{std::abs(this->m_fa)};
402 Scalar abs_fb{std::abs(this->m_fb)};
403 if (this->m_verbose) {
404 this->info(abs_fa < abs_fb ? abs_fa : abs_fb);
405 }
406 if (this->m_converged) {
407 return abs_fa < abs_fb ? this->m_a : this->m_b;
408 }
409
410 this->m_e = this->m_d;
411 this->m_fe = this->m_fd;
412
413 // Takes the double-size secant step
414 Scalar u, fu;
415 if (abs_fa < abs_fb) {
416 u = this->m_a;
417 fu = this->m_fa;
418 } else {
419 u = this->m_b;
420 fu = this->m_fb;
421 }
422 Scalar hba{(this->m_b - this->m_a) / static_cast<Scalar>(2.0)};
423 this->m_c = u - 4.0 * (fu / (this->m_fb - this->m_fa)) * hba;
424 if (std::abs(this->m_c - u) > hba) {
425 this->m_c = this->m_a + hba;
426 }
427 }
428
429 // Call "bracketing" to get a shrinked enclosing interval and to
430 // update the termination criterion
431 this->m_converged =
432 this->bracketing(function) ||
433 (this->m_b - this->m_a) <= this->m_tolerance_bracketing;
434 if (this->m_converged) {
435 return std::abs(this->m_fa) < std::abs(this->m_fb) ? this->m_a
436 : this->m_b;
437 }
438
439 // Determine whether an additional bisection step is needed
440 if ((this->m_b - this->m_a) < this->m_mu * a0_b0) {
441 continue;
442 }
443
444 this->m_e = this->m_d;
445 this->m_fe = this->m_fd;
446
447 // Call "bracketing" to get a shrinked enclosing interval and to
448 // update the termination criterion
449 {
450 Scalar b_a{this->m_b - this->m_a};
451 this->m_c = this->m_a + b_a / 2.0;
452 this->m_converged =
453 this->bracketing(function) ||
454 b_a <= this->m_tolerance_bracketing;
455 }
456 }
457 // Return the approximate root
458 return std::abs(this->m_fa) < std::abs(this->m_fb) ? this->m_a
459 : this->m_b;
460
461#undef CMD
462 }
