54 template <typename FunctionLambda>
55 bool solve_impl(FunctionLambda &&function,
56 const Scalar x_ini,
57 Scalar &x_sol) {
58 this->reset_counters();
59 if (this->m_verbose) {
60 this->header();
61 }
62
63 Scalar a, b, c, fa, fb, fc;
64 if (!this->bracket(function, x_ini, a, b, c, fa, fb, fc)) {
65 if (this->m_verbose) {
66 this->bottom();
67 }
68 x_sol = fa < fc ? a : c;
69 return false;
70 }
71 if (a > c) {
72 std::swap(a, c);
73 std::swap(fa, fc);
74 }
75
76 // x: best point, w: second best, v: previous value of w
77 const Scalar cgold{1 - 1 / this->GOLDEN};
78 Scalar x{b}, w{b}, v{b}, fx{fb}, fw{fb}, fv{fb};
79 Scalar d{0}, e{0};
80 std::string notes{"Initialize"};
81 for (this->m_iterations = 1;
82 this->m_iterations < this->m_max_iterations;
83 ++this->m_iterations) {
84 const Scalar xm{(a + c) / 2};
85 const Scalar tol1{this->m_tolerance * (1 + std::abs(x))};
86 const Scalar tol2{2 * tol1};
87 if (this->m_verbose) {
88 this->info(c - a, notes);
89 }
90 if (std::abs(x - xm) <= tol2 - (c - a) / 2) {
91 this->m_converged = true;
92 break;
93 }
94
95 bool golden{true};
96 if (std::abs(e) > tol1) {
97 // Trial parabolic fit through x, v, w
98 const Scalar r{(x - w) * (fx - fv)};
99 Scalar q{(x - v) * (fx - fw)};
100 Scalar p{(x - v) * q - (x - w) * r};
101 q = 2 * (q - r);
102 if (q > 0) {
103 p = -p;
104 }
105 q = std::abs(q);
106 const Scalar e_old{e};
107 e = d;
108 if (std::abs(p) < std::abs(q * e_old / 2) && p > q * (a - x) &&
109 p < q * (c - x)) {
110 d = p / q;
111 const Scalar u{x + d};
112 if (u - a < tol2 || c - u < tol2) {
113 d = xm >= x ? tol1 : -tol1;
114 }
115 golden = false;
116 notes = "Parabolic";
117 }
118 }
119 if (golden) {
120 e = x >= xm ? a - x : c - x;
121 d = cgold * e;
122 notes = "Golden";
123 }
124
125 const Scalar u{std::abs(d) >= tol1 ? x + d
126 : x + (d >= 0 ? tol1 : -tol1)};
127 const Scalar fu{this->eval(function, u)};
128 if (fu <= fx) {
129 if (u >= x) {
130 a = x;
131 } else {
132 c = x;
133 }
134 v = w;
135 fv = fw;
136 w = x;
137 fw = fx;
138 x = u;
139 fx = fu;
140 } else {
141 if (u < x) {
142 a = u;
143 } else {
144 c = u;
145 }
146 if (fu <= fw || w == x) {
147 v = w;
148 fv = fw;
149 w = u;
150 fw = fu;
151 } else if (fu <= fv || v == x || v == w) {
152 v = u;
153 fv = fu;
154 }
155 }
156 }
157
158 if (this->m_verbose) {
159 this->bottom();
160 }
161 x_sol = x;
162 return this->m_converged;
163 }
