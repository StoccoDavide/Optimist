55 template <typename FunctionLambda>
56 bool solve_impl(FunctionLambda &&function,
57 const Scalar x_ini,
58 Scalar &x_sol) {
59 this->reset_counters();
60 if (this->m_verbose) {
61 this->header();
62 }
63
64 Scalar a, b, c, fa, fb, fc;
65 if (!this->bracket(function, x_ini, a, b, c, fa, fb, fc)) {
66 if (this->m_verbose) {
67 this->bottom();
68 }
69 x_sol = fa < fc ? a : c;
70 return false;
71 }
72 if (a > c) {
73 std::swap(a, c);
74 std::swap(fa, fc);
75 }
76
77 // Two interior points in golden ratio
78 const Scalar r{1 / this->GOLDEN};
79 Scalar x1{c - r * (c - a)}, x2{a + r * (c - a)};
80 Scalar f1{this->eval(function, x1)}, f2{this->eval(function, x2)};
81 for (this->m_iterations = 1;
82 this->m_iterations < this->m_max_iterations;
83 ++this->m_iterations) {
84 const Scalar width{c - a};
85 if (this->m_verbose) {
86 this->info(width);
87 }
88 if (width <= this->m_tolerance * (1 + std::abs(x1) + std::abs(x2))) {
89 this->m_converged = true;
90 break;
91 }
92 if (f1 < f2) {
93 c = x2;
94 x2 = x1;
95 f2 = f1;
96 x1 = c - r * (c - a);
97 f1 = this->eval(function, x1);
98 } else {
99 a = x1;
100 x1 = x2;
101 f1 = f2;
102 x2 = a + r * (c - a);
103 f2 = this->eval(function, x2);
104 }
105 }
106
107 if (this->m_verbose) {
108 this->bottom();
109 }
110 x_sol = f1 < f2 ? x1 : x2;
111 return this->m_converged;
112 }
